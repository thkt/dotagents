// Usage: bun run-followup.ts <copyDir> <mutants.json> <outJsonl>
import { readFileSync, writeFileSync, appendFileSync } from 'node:fs';
import { spawnSync } from 'node:child_process';
const [dir, spec, out] = process.argv.slice(2);
const mutants = Object.fromEntries(JSON.parse(readFileSync(spec, 'utf8')).map((m: any) => [m.id, m]));
const run = (args: string[]) => {
  const t = Date.now();
  const r = spawnSync('bun', ['scripts/tests/runner.ts', 'bun', ...args], { cwd: dir, encoding: 'utf8', maxBuffer: 1 << 28 });
  const text = (r.stdout ?? '') + (r.stderr ?? '');
  const m = text.match(/(\d+) pass[\s\S]*?(\d+) fail/);
  const fails = [...text.matchAll(/^\(fail\) (.*?)(?: \[[\d.]+ms\])?$/gm)].map((x) => x[1]);
  return { code: r.status, pass: m ? +m[1] : null, fail: m ? +m[2] : null, ms: Date.now() - t, fails };
};
const withMutant = (id: string | null, fn: () => unknown) => {
  if (!id) return fn();
  const mu = mutants[id]; const path = `${dir}/${mu.file}`; const orig = readFileSync(path, 'utf8');
  writeFileSync(path, orig.replace(mu.from, mu.to));
  try { return fn(); } finally { writeFileSync(path, orig); }
};
const cp = 'scripts/tests/quality/implement/correction-process.test.ts';
for (const id of [null, 'M5']) for (let i = 0; i < 3; i++)
  appendFileSync(out, JSON.stringify({ kind: 'flake', mutant: id, i, r: withMutant(id, () => run([cp])) }) + '\n');
for (const id of ['M4', 'M6', 'M9'])
  appendFileSync(out, JSON.stringify({ kind: 'assurance', mutant: id, r: withMutant(id, () => run(['scripts/tests/assurance', '--parallel=4'])) }) + '\n');
