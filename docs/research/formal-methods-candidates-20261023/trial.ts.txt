// Independent declarative oracle: required names must occur, every required occurrence must
// succeed, and optional occurrences may only succeed/skip/be neutral. Pending prevents pass.
function ciSpec(values:any[],required:string[]){
  const checks=values.map(v=>({name:v.context,state:v.state}));
  const failed=checks.filter(c=>required.includes(c.name)?!['SUCCESS','PENDING'].includes(c.state):!['SUCCESS','SKIPPED','NEUTRAL','PENDING'].includes(c.state));
  const missing=required.filter(n=>!checks.some(c=>c.name===n));
  const running=checks.filter(c=>c.state==='PENDING');
  const unmet=required.filter(n=>!checks.some(c=>c.name===n)||!checks.filter(c=>c.name===n).every(c=>c.state==='SUCCESS'));
  return {status:failed.length?'failed':missing.length?'missing':running.length?'running':'passed',checks,missing,running,failed,unmet};
}

// Lifecycle oracle uses explicit states and a transition table rather than copied branches.
const symbols=['thread','start','done0','done1','invalidUsage','fail','error','noise','broken','invalidEvent'];
const event={thread:JSON.stringify({type:'thread.started',thread_id:'t'}),start:JSON.stringify({type:'turn.started'}),done0:JSON.stringify({type:'turn.completed',usage:zero}),done1:JSON.stringify({type:'turn.completed',usage:usage()}),invalidUsage:JSON.stringify({type:'turn.completed',usage:usage(0,1,0)}),fail:JSON.stringify({type:'turn.failed'}),error:JSON.stringify({type:'error'}),noise:JSON.stringify({type:'item.completed'}),broken:'{',invalidEvent:'{}'};
function lifecycleSpec(trace:string[]){
  let state='unidentified',threadId:string|null=null,ordinal=0,problem=false;const turns:any[]=[];
  const transitions:any={unidentified:{thread:'idle'},idle:{start:'active'},active:{done0:'idle',done1:'idle',fail:'idle'}};
  for(const [index,s]of trace.entries()){
    if(s==='noise')continue;
    if(s==='error'){problem=true;continue;}
    const next=transitions[state]?.[s];
    if(!next){problem=true;break;}
    if(s==='thread')threadId='t';
    if(s==='start')ordinal++;
    if(s==='fail')problem=true;
    if(s.startsWith('done'))turns.push({ordinal,line:index+1,usage:s==='done0'?zero:usage()});
    state=next;
  }
  if(!threadId||state==='active'||turns.length===0)problem=true;
  const observed=turns.length?usage(turns.reduce((n,t)=>n+t.usage.input_tokens,0),0,turns.reduce((n,t)=>n+t.usage.output_tokens,0)):null;
  return {threadId,turns,observed,problem};
}

}
